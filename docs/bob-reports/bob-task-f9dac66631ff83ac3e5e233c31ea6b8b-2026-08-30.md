> 🔧 **write_file** — """Parse Rust cargo test output into structured results.

Supports:
- cargo test summary line (ok and FAILED variants)

Examples:
    test result: ok. 5 passed; 0 failed; 0 ignored; 0 measured; 0 filtered out
    test result: FAILED. 3 passed; 2 failed; 1 ignored; 0 measured; 0 filtered out
"""

from __future__ import annotations

import re

from releaseguard.models.core import TestRunResult, TestFailure


# cargo test emits one or more result lines of the form:
#   "test result: ok. 5 passed; 0 failed; 0 ignored; 0 measured; 0 filtered out"
# When multiple test binaries run there may be several such lines; we aggregate.
_RESULT_RE = re.compile(
    r"test result:\s+(?:ok|FAILED|error)\.\s+"
    r"(?P<passed>\d+)\s+passed;\s+"
    r"(?P<failed>\d+)\s+failed;\s+"
    r"(?P<ignored>\d+)\s+ignored",
    re.IGNORECASE,
)

# Individual failing test: "FAILED src::module::test_name"
_FAILED_TEST_RE = re.compile(r"^FAILED\s+(\S+)", re.MULTILINE)

# Individual test failure lines in verbose output:
# "test some::path::test_name ... FAILED"
_VERBOSE_FAIL_RE = re.compile(r"^test\s+(\S+)\s+\.\.\.\s+FAILED", re.MULTILINE)


def parse_rust(result: TestRunResult) -> TestRunResult:
    """Populate *result* with parsed counts from cargo test output.

    Aggregates counts across all test binaries when a workspace has multiple
    crates.  Mutates *result* in-place and returns it.
    """
    output = result.stdout + "\n" + result.stderr

    total_passed = 0
    total_failed = 0
    total_ignored = 0
    found_any = False

    for m in _RESULT_RE.finditer(output):
        found_any = True
        total_passed += int(m.group("passed"))
        total_failed += int(m.group("failed"))
        total_ignored += int(m.group("ignored"))

    if not found_any:
        # Unrecognised output — leave counts as None
        return result

    result.passed = total_passed
    result.failed = total_failed
    result.skipped = total_ignored
    result.total = total_passed + total_failed + total_ignored

    # Collect failing test names from verbose output
    if result.failed:
        failed_names: list[str] = _VERBOSE_FAIL_RE.findall(output)
        if not failed_names:
            failed_names = _FAILED_TEST_RE.findall(output)
        result.failures = [TestFailure(name=n) for n in failed_names]

    return result


> 🔧 **write_file** — """Parse Java (Maven / Gradle) test output into structured results.

Supports:
- Maven Surefire / Failsafe summary lines
- Gradle test summary lines

Maven examples:
    Tests run: 10, Failures: 1, Errors: 0, Skipped: 2
    Tests run: 5, Failures: 0, Errors: 0, Skipped: 0

Gradle examples:
    10 tests completed, 1 failed
    10 tests completed, 1 failed, 2 skipped
    5 tests completed
"""

from __future__ import annotations

import re

from releaseguard.models.core import TestRunResult, TestFailure


# ---------------------------------------------------------------------------
# Maven Surefire
# Matches lines like:
#   Tests run: 10, Failures: 1, Errors: 0, Skipped: 2
# ---------------------------------------------------------------------------

_MAVEN_RE = re.compile(
    r"Tests run:\s*(?P<run>\d+)"
    r",\s*Failures:\s*(?P<failures>\d+)"
    r",\s*Errors:\s*(?P<errors>\d+)"
    r",\s*Skipped:\s*(?P<skipped>\d+)",
    re.IGNORECASE,
)

# Individual failing test line from Maven Surefire verbose output:
#   FAILED com.example.FooTest#testSomething
_MAVEN_FAILED_TEST_RE = re.compile(r"FAILED\s+([\w.#]+)", re.MULTILINE)

# ---------------------------------------------------------------------------
# Gradle
# Matches lines like:
#   10 tests completed, 1 failed
#   10 tests completed, 1 failed, 2 skipped
#   5 tests completed
# ---------------------------------------------------------------------------

_GRADLE_RE = re.compile(
    r"(?P<total>\d+)\s+tests?\s+completed"
    r"(?:,\s*(?P<failed>\d+)\s+failed)?"
    r"(?:,\s*(?P<skipped>\d+)\s+skipped)?",
    re.IGNORECASE,
)


def parse_java(result: TestRunResult) -> TestRunResult:
    """Populate *result* with parsed counts from Maven/Gradle test output.

    Aggregates counts across all Maven module lines when present.
    Mutates *result* in-place and returns it.
    """
    output = result.stdout + "\n" + result.stderr

    if _try_maven(result, output):
        return result
    if _try_gradle(result, output):
        return result

    # Unrecognised output — leave counts as None
    return result


def _try_maven(result: TestRunResult, output: str) -> bool:
    """Parse Maven Surefire output. Returns True if counts found."""
    matches = list(_MAVEN_RE.finditer(output))
    if not matches:
        return False

    total_run = 0
    total_failures = 0
    total_errors = 0
    total_skipped = 0

    for m in matches:
        total_run += int(m.group("run"))
        total_failures += int(m.group("failures"))
        total_errors += int(m.group("errors"))
        total_skipped += int(m.group("skipped"))

    result.passed = total_run - total_failures - total_errors - total_skipped
    result.failed = total_failures + total_errors
    result.skipped = total_skipped
    result.total = total_run

    # Ensure passed is never negative due to rounding in aggregated output
    if result.passed < 0:
        result.passed = 0

    if result.failed:
        failed_names = _MAVEN_FAILED_TEST_RE.findall(output)
        result.failures = [TestFailure(name=n) for n in failed_names]

    return True


def _try_gradle(result: TestRunResult, output: str) -> bool:
    """Parse Gradle test summary. Returns True if counts found."""
    m = _GRADLE_RE.search(output)
    if not m:
        return False

    total = int(m.group("total"))
    failed = int(m.group("failed")) if m.group("failed") else 0
    skipped = int(m.group("skipped")) if m.group("skipped") else 0
    passed = total - failed - skipped

    result.total = total
    result.failed = failed
    result.skipped = skipped
    result.passed = max(passed, 0)
    return True


> 🔧 **write_file** — """Parse Go test output into structured results.

Supports:
- go test verbose output (--- PASS / --- FAIL per-test lines)
- go test non-verbose summary (PASS / FAIL exit line)
- go test -json output (partial: counts from action fields)

Examples (verbose):
    --- PASS: TestSomething (0.00s)
    --- FAIL: TestBroken (0.01s)
    PASS
    ok      mymodule    0.005s

Examples (non-verbose, compile/no tests):
    FAIL    mymodule [build failed]
    ok      mymodule    (cached)
"""

from __future__ import annotations

import re

from releaseguard.models.core import TestRunResult, TestFailure


# Individual test outcome lines in verbose mode:
#   --- PASS: TestFunctionName (0.00s)
#   --- FAIL: TestFunctionName (0.01s)
_PASS_LINE_RE = re.compile(r"^--- PASS:\s+(\S+)", re.MULTILINE)
_FAIL_LINE_RE = re.compile(r"^--- FAIL:\s+(\S+)", re.MULTILINE)

# go test -json emits lines like {"Action":"pass","Test":"TestFoo",...}
# We detect JSON mode by the presence of these structured lines.
_JSON_ACTION_RE = re.compile(
    r'\{"Action":"(?P<action>pass|fail|skip)","Test":"(?P<test>[^"]+)"',
)

# Package-level result line (not a test name):
#   ok      github.com/foo/bar   0.005s
#   FAIL    github.com/foo/bar   0.123s
_PKG_OK_RE = re.compile(r"^ok\s+\S+", re.MULTILINE)
_PKG_FAIL_RE = re.compile(r"^FAIL\s+\S+", re.MULTILINE)


def parse_go(result: TestRunResult) -> TestRunResult:
    """Populate *result* with parsed counts from go test output.

    Mutates *result* in-place and returns it.
    """
    output = result.stdout + "\n" + result.stderr

    # Try JSON output first (go test -json)
    if _try_go_json(result, output):
        return result

    # Try verbose output (--- PASS / --- FAIL lines)
    if _try_go_verbose(result, output):
        return result

    # Fall back to package-level pass/fail only
    _try_go_summary(result, output)
    return result


def _try_go_json(result: TestRunResult, output: str) -> bool:
    """Parse go test -json output. Returns True if test-level data found."""
    matches = list(_JSON_ACTION_RE.finditer(output))
    if not matches:
        return False

    passed = 0
    failed = 0
    skipped = 0
    failures: list[str] = []

    for m in matches:
        action = m.group("action")
        test = m.group("test")
        if action == "pass":
            passed += 1
        elif action == "fail":
            failed += 1
            failures.append(test)
        elif action == "skip":
            skipped += 1

    result.passed = passed
    result.failed = failed
    result.skipped = skipped
    result.total = passed + failed + skipped
    result.failures = [TestFailure(name=n) for n in failures]
    return True


def _try_go_verbose(result: TestRunResult, output: str) -> bool:
    """Parse verbose go test output. Returns True if per-test lines found."""
    pass_names = _PASS_LINE_RE.findall(output)
    fail_names = _FAIL_LINE_RE.findall(output)
    if not pass_names and not fail_names:
        return False

    result.passed = len(pass_names)
    result.failed = len(fail_names)
    result.skipped = 0
    result.total = result.passed + result.failed
    result.failures = [TestFailure(name=n) for n in fail_names]
    return True


def _try_go_summary(result: TestRunResult, output: str) -> None:
    """Record pass/fail at package level when no per-test data is available.

    When only package-level outcome is known, we know whether the suite
    passed or failed but not individual test counts — leave total/passed/
    failed as None to avoid silently reporting 0 tests.
    """
    # We only set execution_error when we know the suite failed but have no
    # per-test data to report — this prevents "0 tests passed" appearing.
    pkg_failures = _PKG_FAIL_RE.findall(output)
    if pkg_failures and result.failed is None:
        result.execution_error = (
            f"go test reported package failure(s): {', '.join(pkg_failures)}"
        )

    Language.JAVA: parse_java,

def collect_evidence(results: list[TestRunResult]) -> list[TestRunResult]:
    """Parse raw test output for each result and return enriched results."""
    enriched = []
    for result in results:
        parser = _PARSERS.get(result.project.language)
        if parser is not None and result.tooling_available:
            result = parser(result)
        enriched.append(result)
    return enriched

### How counts are extracted

- **Node.js**: three tried in order — built-in regex on the `ℹ`/`:information_source:` lines → Jest `Tests:` line → Mocha `N passing`/`N failing` lines. First match wins.
- **Rust**: `_RESULT_RE` matches each `test result:` line; all matching lines are summed (workspace support). `ignored` maps to `skipped`.
- **Go**: tries JSON `"Action":"pass|fail|skip"` objects first, then `--- PASS/FAIL:` verbose lines, then package-level `FAIL` marker (sets `execution_error`, leaves counts `None`).
- **Java**: Maven `Tests run:` lines are found and summed; `Failures + Errors` → `failed`, `run - failed - skipped` → `passed`. Gradle `N tests completed` is parsed directly.
- **Unknown output / no match**: counts are left as `None` — never silently reported as `0 tests`.
